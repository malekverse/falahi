# DATA_MODELS.md — Database Schema (Canonical Source)
## Filahi — Supabase / PostgreSQL

> This file is the single source of truth for all database tables.
> Never modify tables directly in the Supabase dashboard — always write a migration file in `supabase/migrations/`.
> All prices are stored as INTEGER in millimes (1 TND = 1000 millimes).

## Table: profiles

Linked 1:1 to Supabase Auth `auth.users`. Every authenticated user has exactly one profile.

```sql
CREATE TABLE profiles (
  id              UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name       TEXT NOT NULL,
  phone_number    TEXT UNIQUE NOT NULL,         -- E.164 format: +21620xxxxxx
  role            TEXT NOT NULL CHECK (role IN ('farmer', 'driver', 'courier', 'buyer', 'admin')),
  whatsapp_id     TEXT UNIQUE,                  -- Meta WA sender ID (for farmer bot linking)
  avatar_url      TEXT,
  preferred_lang  TEXT DEFAULT 'ar' CHECK (preferred_lang IN ('ar', 'fr')),
  is_active       BOOLEAN DEFAULT TRUE,
  created_at      TIMESTAMPTZ DEFAULT NOW(),
  updated_at      TIMESTAMPTZ DEFAULT NOW()
);

-- RLS
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users read own profile"
  ON profiles FOR SELECT USING (auth.uid() = id);

CREATE POLICY "Users update own profile"
  ON profiles FOR UPDATE USING (auth.uid() = id);

CREATE POLICY "Admin reads all profiles"
  ON profiles FOR SELECT USING (
    EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'admin')
  );
```

## Table: inventory_items

Every produce listing, created by the farmer bot or manually by admin.

```sql
CREATE TABLE inventory_items (
  id                      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  farmer_id               UUID NOT NULL REFERENCES profiles(id),
  product_name            TEXT NOT NULL,
  product_category        TEXT NOT NULL CHECK (product_category IN (
                            'vegetables', 'fruit', 'eggs', 'honey',
                            'olive_oil', 'legumes', 'grains', 'herbs', 'other'
                          )),
  quantity                NUMERIC NOT NULL CHECK (quantity > 0),
  unit                    TEXT NOT NULL CHECK (unit IN ('kg', 'hara', 'litra', 'crate', 'piece', 'ton')),
  asking_price_millimes   INT NOT NULL CHECK (asking_price_millimes > 0),  -- Per unit
  platform_price_millimes INT,                   -- Computed: asking_price * 1.12 (set on insert by trigger)
  location_name           TEXT NOT NULL,
  coordinates             GEOMETRY(Point, 4326),
  harvest_date            DATE,
  shelf_life_days         INT,                   -- Set by admin or AI based on product type
  expires_at              TIMESTAMPTZ,           -- Auto-computed: harvest_date + shelf_life_days
  status                  TEXT NOT NULL DEFAULT 'pending_confirmation'
                            CHECK (status IN ('pending_confirmation', 'available', 'reserved', 'sold', 'expired')),
  source                  TEXT DEFAULT 'whatsapp_bot' CHECK (source IN ('whatsapp_bot', 'admin', 'web')),
  raw_transcription       TEXT,                  -- Original Darija text from Whisper
  ai_confidence_score     NUMERIC(3,2),          -- 0.00 to 1.00
  whatsapp_message_id     TEXT UNIQUE,           -- For idempotency
  notes                   TEXT,
  created_at              TIMESTAMPTZ DEFAULT NOW(),
  updated_at              TIMESTAMPTZ DEFAULT NOW()
);

-- Spatial index (mandatory for performance)
CREATE INDEX idx_inventory_coordinates ON inventory_items USING GIST (coordinates);
CREATE INDEX idx_inventory_status ON inventory_items (status);
CREATE INDEX idx_inventory_farmer ON inventory_items (farmer_id);

-- Trigger: auto-compute platform price
CREATE OR REPLACE FUNCTION compute_platform_price()
RETURNS TRIGGER AS $$
BEGIN
  NEW.platform_price_millimes := ROUND(NEW.asking_price_millimes * 1.12);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_platform_price
  BEFORE INSERT OR UPDATE OF asking_price_millimes ON inventory_items
  FOR EACH ROW EXECUTE FUNCTION compute_platform_price();

ALTER TABLE inventory_items ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Anyone can view available inventory"
  ON inventory_items FOR SELECT USING (status = 'available');

CREATE POLICY "Farmers manage own inventory"
  ON inventory_items FOR ALL USING (auth.uid() = farmer_id);

CREATE POLICY "Admin manages all inventory"
  ON inventory_items FOR ALL USING (
    EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'admin')
  );
```

## Table: trips

One trip per long-haul run from farm region to micro-hub.

```sql
CREATE TABLE trips (
  id                      UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  driver_id               UUID REFERENCES driver_profiles(id),
  order_ids               UUID[],                -- Array of orders in this batch
  origin_location_name    TEXT NOT NULL,
  origin_coordinates      GEOMETRY(Point, 4326),
  hub_id                  UUID REFERENCES hubs(id),
  status                  TEXT NOT NULL DEFAULT 'pending'
                            CHECK (status IN (
                              'pending', 'accepted', 'in_transit',
                              'arrived_hub', 'delivered', 'settled', 'disputed'
                            )),
  otp_pickup              TEXT NOT NULL,          -- 4-digit code, sent to farmer
  otp_delivery            TEXT NOT NULL,          -- 4-digit code, sent to hub manager
  cargo_value_millimes    INT NOT NULL,           -- Total value of cargo for tier check
  driver_fee_millimes     INT NOT NULL,           -- What driver earns
  platform_fee_millimes   INT NOT NULL,
  last_known_lat          NUMERIC(10, 7),
  last_known_lng          NUMERIC(10, 7),
  last_ping_at            TIMESTAMPTZ,
  estimated_arrival_at    TIMESTAMPTZ,
  accepted_at             TIMESTAMPTZ,
  picked_up_at            TIMESTAMPTZ,
  delivered_at            TIMESTAMPTZ,
  settled_at              TIMESTAMPTZ,
  created_at              TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_trips_status ON trips (status);
CREATE INDEX idx_trips_driver ON trips (driver_id);

-- Atomic OTP validation RPC (never validate OTP in application layer)
CREATE OR REPLACE FUNCTION validate_pickup_otp(trip_id UUID, otp_input TEXT)
RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_trip trips%ROWTYPE;
BEGIN
  SELECT * INTO v_trip FROM trips WHERE id = trip_id FOR UPDATE;

  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'error', 'trip_not_found');
  END IF;

  IF v_trip.status != 'accepted' THEN
    RETURN jsonb_build_object('success', false, 'error', 'invalid_status');
  END IF;

  IF v_trip.otp_pickup != otp_input THEN
    RETURN jsonb_build_object('success', false, 'error', 'invalid_otp');
  END IF;

  UPDATE trips SET status = 'in_transit', picked_up_at = NOW() WHERE id = trip_id;

  RETURN jsonb_build_object('success', true);
END;
$$;

CREATE OR REPLACE FUNCTION validate_delivery_otp(trip_id UUID, otp_input TEXT)
RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_trip trips%ROWTYPE;
BEGIN
  SELECT * INTO v_trip FROM trips WHERE id = trip_id FOR UPDATE;

  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'error', 'trip_not_found');
  END IF;

  IF v_trip.status != 'in_transit' THEN
    RETURN jsonb_build_object('success', false, 'error', 'invalid_status');
  END IF;

  IF v_trip.otp_delivery != otp_input THEN
    RETURN jsonb_build_object('success', false, 'error', 'invalid_otp');
  END IF;

  UPDATE trips SET
    status = 'delivered',
    delivered_at = NOW()
  WHERE id = trip_id;

  RETURN jsonb_build_object('success', true);
END;
$$;

ALTER TABLE trips ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Drivers read own trips"
  ON trips FOR SELECT USING (
    EXISTS (SELECT 1 FROM driver_profiles WHERE id = auth.uid() AND id = driver_id)
  );

CREATE POLICY "Admin manages all trips"
  ON trips FOR ALL USING (
    EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'admin')
  );
```

## Table: trip_location_cache

Stores GPS pings when driver is offline. Flushed to Realtime on reconnect.

```sql
CREATE TABLE trip_location_cache (
  id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  trip_id     UUID NOT NULL REFERENCES trips(id) ON DELETE CASCADE,
  lat         NUMERIC(10, 7) NOT NULL,
  lng         NUMERIC(10, 7) NOT NULL,
  recorded_at TIMESTAMPTZ NOT NULL,              -- Device timestamp
  synced_at   TIMESTAMPTZ DEFAULT NOW()          -- Server receipt time
);

CREATE INDEX idx_location_cache_trip ON trip_location_cache (trip_id, recorded_at);

-- Cleanup old cache entries after 48 hours
CREATE OR REPLACE FUNCTION cleanup_old_location_cache()
RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  DELETE FROM trip_location_cache WHERE synced_at < NOW() - INTERVAL '48 hours';
END;
$$;
```

## Table: ledger_entries

Immutable financial event log. Never update or delete rows here.

```sql
CREATE TABLE ledger_entries (
  id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  trip_id             UUID REFERENCES trips(id),
  order_id            UUID REFERENCES orders(id),
  entry_type          TEXT NOT NULL CHECK (entry_type IN (
                        'buyer_payment', 'platform_commission',
                        'driver_fee', 'farmer_payout', 'refund', 'adjustment'
                      )),
  amount_millimes     INT NOT NULL,              -- Positive = credit, Negative = debit
  currency            TEXT DEFAULT 'TND',
  party_id            UUID REFERENCES profiles(id),
  payment_method      TEXT CHECK (payment_method IN ('cod', 'flouci', 'walletii', 'bank', 'manual')),
  reference           TEXT,                      -- External reference (e.g., Flouci transaction ID)
  notes               TEXT,
  created_at          TIMESTAMPTZ DEFAULT NOW()  -- NEVER add updated_at to ledger
);

-- Ledger is append-only: no UPDATE or DELETE
ALTER TABLE ledger_entries ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Admin reads ledger"
  ON ledger_entries FOR SELECT USING (
    EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'admin')
  );

CREATE POLICY "Ledger is append-only via server"
  ON ledger_entries FOR INSERT WITH CHECK (TRUE);
```
